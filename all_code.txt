class ProfitAssumptionsRead(BaseModel):
    supplier_cost_change_pct: Decimal
    selling_price_change_pct: Decimal
    volume_change_pct: Decimal


class ProfitAssumptions(ProfitAssumptionsRead):
    supplier_cost_change_pct: Decimal = Field(
        ge=engine.SUPPLIER_COST_CHANGE_MIN,
        le=engine.SUPPLIER_COST_CHANGE_MAX,
    )
    selling_price_change_pct: Decimal = Field(
        ge=engine.SELLING_PRICE_CHANGE_MIN,
        le=engine.SELLING_PRICE_CHANGE_MAX,
    )
    volume_change_pct: Decimal = Field(
        ge=engine.VOLUME_CHANGE_MIN,
        le=engine.VOLUME_CHANGE_MAX,
    )

    model_config = {"extra": "forbid"}

    @field_validator(
        "supplier_cost_change_pct",
        "selling_price_change_pct",
        "volume_change_pct",
    )
    @classmethod
    def _finite(cls, value: Decimal) -> Decimal:
        if not value.is_finite():
            raise ValueError("assumptions must be finite numbers")
        return value

    scenario_type: ScenarioType = engine.SCENARIO_TYPE
    source_id: uuid.UUID

    name: str = Field(min_length=1, max_length=200)

    def _name_not_blank(cls, value: str) -> str:
        cleaned = value.strip()
        if not cleaned:
            raise ValueError("name must not be blank")
        return cleaned

"""
Scenario Lab API.

Contract notes:
  - The tenant always comes from the authenticated user, never from the
    request body. ``source_id`` is verified against the tenant inside the
    baseline resolver.
  - Preview (`POST /preview`) is the cheap, read-only operation; save
    (`POST ""`) persists. Both run the same engine on the same baseline,
    so a preview and a saved scenario with equal inputs agree.
  - Missing prerequisites return 422 with a structured body naming the
    missing concept(s): the UI should explain, not just fail.
"""

from __future__ import annotations

import uuid
from datetime import UTC, datetime
from typing import Annotated

from fastapi import APIRouter, Depends, HTTPException, status
from sqlalchemy.ext.asyncio import AsyncSession

from app.api.deps import CurrentTenantId, require_permission
from app.db.models.user import User
from app.db.session import get_db
from app.schemas.scenarios import (
    ScenarioCalculateRequest,
    ScenarioCapabilitiesRead,
    ScenarioCompareRequest,
    ScenarioCompareResponse,
    ScenarioPreviewResponse,
    ScenarioRead,
    ScenarioSaveRequest,
    ScenarioSummaryRead,

)
from app.services.scenarios import engine

router = APIRouter(prefix="/scenarios", tags=["scenarios"])


def _baseline_error(exc: BaselineUnavailable) -> HTTPException:
    status_code = (
        status.HTTP_404_NOT_FOUND
        if exc.code == "source_not_found"
        else status.HTTP_422_UNPROCESSABLE_ENTITY
    )
    return HTTPException(
        status_code=status_code,
        detail={
            "code": exc.code,
            "message": exc.detail,
            "missing_concepts": exc.missing,
        },
    )


@router.get("/capabilities", response_model=ScenarioCapabilitiesRead)
async def scenario_capabilities(
    db: Annotated[AsyncSession, Depends(get_db)],
    tenant_id: CurrentTenantId,
    _user: Annotated[User, Depends(require_permission("dataset:read"))],
) -> ScenarioCapabilitiesRead:
    """
    Which sources can support profit sensitivity, and why others cannot.
    Computed from confirmed revenue, COGS, and currency mappings.
    """
    sources: list[SourceCapabilityRead] = await scenario_service.capabilities(
        db, tenant_id=tenant_id
    )
    return ScenarioCapabilitiesRead(
        scenario_type=engine.SCENARIO_TYPE,
        formula_version=engine.FORMULA_VERSION,
        assumption_bounds={
            "supplier_cost_change_pct": {
                "min": engine.SUPPLIER_COST_CHANGE_MIN,
                "max": engine.SUPPLIER_COST_CHANGE_MAX,
            },
            "selling_price_change_pct": {
                "min": engine.SELLING_PRICE_CHANGE_MIN,
                "max": engine.SELLING_PRICE_CHANGE_MAX,
            },
            "volume_change_pct": {
                "min": engine.VOLUME_CHANGE_MIN,
                "max": engine.VOLUME_CHANGE_MAX,
            },
        },
        supported=any(s.supported for s in sources),
        sources=sources,
        limitations=engine.default_limitations(),
    )


@router.post("/preview", response_model=ScenarioPreviewResponse)
async def preview_scenario(
    body: ScenarioCalculateRequest,
    db: Annotated[AsyncSession, Depends(get_db)],
    tenant_id: CurrentTenantId,
    _user: Annotated[User, Depends(require_permission("dataset:read"))],
) -> ScenarioPreviewResponse:
    """Calculate without saving. Read-only and side-effect free."""
    try:
        result_document, baseline, _result = await scenario_service.calculate(
            db,
            tenant_id=tenant_id,
            source_id=body.source_id,
            assumptions=engine.ProfitAssumptions(
                supplier_cost_change_pct=body.assumptions.supplier_cost_change_pct,
                selling_price_change_pct=body.assumptions.selling_price_change_pct,
                volume_change_pct=body.assumptions.volume_change_pct,
            ),
        )
    except BaselineUnavailable as exc:
        raise _baseline_error(exc) from exc
    except scenario_service.ScenarioServiceError as exc:
        raise HTTPException(status_code=422, detail=str(exc)) from exc

    return ScenarioPreviewResponse(
        result=scenario_service.result_read(result_document),
        provenance=scenario_service.provenance_read(
            {
                **baseline.to_snapshot(),
                "calculated_at": datetime.now(UTC).isoformat(),
            }
        ),
        calculated_at=datetime.now(UTC),
    )

@router.post("/{scenario_id}/recalculate", response_model=ScenarioRead)
async def recalculate_scenario(
    scenario_id: uuid.UUID,
    db: Annotated[AsyncSession, Depends(get_db)],
    tenant_id: CurrentTenantId,
    user: Annotated[User, Depends(require_permission("dataset:write"))],
) -> ScenarioRead:
    """
    Re-run against the current baseline using the saved assumptions.
    If the source can no longer support the scenario, the run is marked
    stale and the previous numbers are left intact.
    """
    try:
        scenario = await scenario_service.recalculate(
            db, tenant_id=tenant_id, scenario_id=scenario_id, user_id=user.id
        )
    except scenario_service.ScenarioNotFound as exc:
        raise HTTPException(status_code=404, detail="scenario not found") from exc
    except BaselineUnavailable as exc:
        await db.commit()
        raise _baseline_error(exc) from exc
    except scenario_service.ScenarioServiceError as exc:
        raise HTTPException(status_code=422, detail=str(exc)) from exc
    await db.commit()
    await db.refresh(scenario)
    return scenario_service.to_read(scenario)

    """
    Flag saved scenarios whose source data has moved on since they were

    """
    updated = await scenario_service.mark_stale_scenarios(db, tenant_id=tenant_id)
    await db.commit()
    return {"marked_stale": updated}

"""
Deterministic procurement and gross-profit sensitivity calculations.

Pure functions only: no database, filesystem, network, or LLM access.
"""

from __future__ import annotations

from dataclasses import dataclass, field
from decimal import ROUND_HALF_UP, Decimal, localcontext

SCENARIO_TYPE = "profit_sensitivity_v1"
FORMULA_VERSION = 1

SUPPLIER_COST_CHANGE_MIN = Decimal("0")
SUPPLIER_COST_CHANGE_MAX = Decimal("30")
SELLING_PRICE_CHANGE_MIN = Decimal("-10")
SELLING_PRICE_CHANGE_MAX = Decimal("20")
VOLUME_CHANGE_MIN = Decimal("-30")
VOLUME_CHANGE_MAX = Decimal("30")

_MONEY_QUANT = Decimal("0.01")
_PCT_QUANT = Decimal("0.01")


class ScenarioCalculationError(Exception):
    """Inputs are individually well-formed but the calculation is undefined."""


@dataclass(frozen=True)
class ProfitBaseline:
    revenue: Decimal
    cogs: Decimal
    currency: str


@dataclass(frozen=True)
class ProfitAssumptions:
    supplier_cost_change_pct: Decimal
    selling_price_change_pct: Decimal
    volume_change_pct: Decimal


@dataclass
class ProfitSensitivityResult:
    scenario_type: str
    formula_version: int
    currency: str
    assumptions: ProfitAssumptions
    baseline_revenue: Decimal
    baseline_cogs: Decimal
    baseline_gross_profit: Decimal
    baseline_gross_margin_pct: Decimal | None
    scenario_revenue: Decimal
    scenario_cogs: Decimal
    scenario_gross_profit: Decimal
    scenario_gross_margin_pct: Decimal | None
    gross_profit_change: Decimal
    gross_profit_change_pct: Decimal | None
    gross_margin_change_percentage_points: Decimal | None
    explanation: str
    limitations: list[str] = field(default_factory=list)


def _rounded(value: Decimal, quant: Decimal) -> Decimal:
    return value.quantize(quant, rounding=ROUND_HALF_UP)


def _require_finite(value: Decimal, field_name: str) -> None:
    if not value.is_finite():
        raise ScenarioCalculationError(f"{field_name} must be a finite number")


def _validate_assumption(value: Decimal, *, name: str, minimum: Decimal, maximum: Decimal) -> None:
    _require_finite(value, name)
    if not minimum <= value <= maximum:
        raise ScenarioCalculationError(f"{name} must be between {minimum} and {maximum} percent")


def simulate_profit(
    *,
    baseline: ProfitBaseline,
    assumptions: ProfitAssumptions,
) -> ProfitSensitivityResult:
    """
    Calculate the gross-profit impact while holding the purchasing mix fixed.

    Both revenue and COGS scale with volume. Selling price affects revenue;
    supplier cost affects COGS. This is sensitivity analysis, not a forecast.
    """
    _require_finite(baseline.revenue, "baseline.revenue")
    _require_finite(baseline.cogs, "baseline.cogs")
    if baseline.revenue < 0:
        raise ScenarioCalculationError("baseline revenue must be non-negative")
    if baseline.cogs < 0:
        raise ScenarioCalculationError("baseline COGS must be non-negative")
    if not baseline.currency or baseline.currency != baseline.currency.strip():
        raise ScenarioCalculationError("baseline currency must be a normalized code")

    _validate_assumption(
        assumptions.supplier_cost_change_pct,
        name="supplier_cost_change_pct",
        minimum=SUPPLIER_COST_CHANGE_MIN,
        maximum=SUPPLIER_COST_CHANGE_MAX,
    )
    _validate_assumption(
        assumptions.selling_price_change_pct,
        name="selling_price_change_pct",
        minimum=SELLING_PRICE_CHANGE_MIN,
        maximum=SELLING_PRICE_CHANGE_MAX,
    )
    _validate_assumption(
        assumptions.volume_change_pct,
        name="volume_change_pct",
        minimum=VOLUME_CHANGE_MIN,
        maximum=VOLUME_CHANGE_MAX,
    )

    with localcontext() as context:
        context.prec = 40
        volume_factor = Decimal(1) + assumptions.volume_change_pct / Decimal(100)
        revenue_factor = Decimal(1) + assumptions.selling_price_change_pct / Decimal(100)
        cogs_factor = Decimal(1) + assumptions.supplier_cost_change_pct / Decimal(100)

        baseline_revenue = _rounded(baseline.revenue, _MONEY_QUANT)
        baseline_cogs = _rounded(baseline.cogs, _MONEY_QUANT)
        baseline_profit = _rounded(baseline_revenue - baseline_cogs, _MONEY_QUANT)
        scenario_revenue = _rounded(baseline.revenue * revenue_factor * volume_factor, _MONEY_QUANT)
        scenario_cogs = _rounded(baseline.cogs * cogs_factor * volume_factor, _MONEY_QUANT)
        scenario_profit = _rounded(scenario_revenue - scenario_cogs, _MONEY_QUANT)
        profit_change = _rounded(scenario_profit - baseline_profit, _MONEY_QUANT)

        baseline_margin = (
            _rounded(baseline_profit / baseline_revenue * Decimal(100), _PCT_QUANT)
            if baseline_revenue > 0
            else None
        )
        scenario_margin = (
            _rounded(scenario_profit / scenario_revenue * Decimal(100), _PCT_QUANT)
            if scenario_revenue > 0
            else None
        )
        profit_change_pct = (
            _rounded(profit_change / abs(baseline_profit) * Decimal(100), _PCT_QUANT)
            if baseline_profit != 0
            else None
        )
        margin_change = (
            _rounded(scenario_margin - baseline_margin, _PCT_QUANT)
            if scenario_margin is not None and baseline_margin is not None
            else None
        )

    result = ProfitSensitivityResult(
        scenario_type=SCENARIO_TYPE,
        formula_version=FORMULA_VERSION,
        currency=baseline.currency,
        assumptions=assumptions,
        baseline_revenue=baseline_revenue,
        baseline_cogs=baseline_cogs,
        baseline_gross_profit=baseline_profit,
        baseline_gross_margin_pct=baseline_margin,
        scenario_revenue=scenario_revenue,
        scenario_cogs=scenario_cogs,
        scenario_gross_profit=scenario_profit,
        scenario_gross_margin_pct=scenario_margin,
        gross_profit_change=profit_change,
        gross_profit_change_pct=profit_change_pct,
        gross_margin_change_percentage_points=margin_change,
        explanation="",
        limitations=default_limitations(),
    )
    result.explanation = explain_deterministically(result)
    return result


def default_limitations() -> list[str]:
    return [
        "Sales volume changes proportionally scale both revenue and COGS.",
        "Selling-price changes affect revenue only; supplier-cost changes affect COGS only.",
        "The purchasing mix and unit volumes are otherwise held constant.",
        "This is deterministic sensitivity analysis, not a demand or profit forecast.",
        "Demand elasticity, taxes, discounts, fixed costs, inventory constraints, and currency conversion are not modeled.",
    ]


def explain_deterministically(result: ProfitSensitivityResult) -> str:
    """Render a concise explanation from calculated values without an LLM."""
    change_label = "increase" if result.gross_profit_change >= 0 else "decrease"
    text = (
        f"Under the supplied assumptions, gross profit would {change_label} by "
        f"{result.currency} {abs(result.gross_profit_change):,.2f}, from "
        f"{result.currency} {result.baseline_gross_profit:,.2f} to "
        f"{result.currency} {result.scenario_gross_profit:,.2f}."
    )
    if result.baseline_gross_profit == 0:
        text += " A percentage change is undefined because baseline gross profit is zero."
    elif result.gross_profit_change_pct is not None:
        text += f" That is a {result.gross_profit_change_pct:+}% change in gross profit."
    if result.gross_margin_change_percentage_points is not None:
        text += (
            f" Gross margin moves from {result.baseline_gross_margin_pct}% to "
            f"{result.scenario_gross_margin_pct}% "
            f"({result.gross_margin_change_percentage_points:+} percentage points)."
        )
    else:
        text += " Gross margin is undefined because scenario revenue is zero."
    return text

"""
Resolve a tenant-owned, auditable revenue/COGS baseline for profit analysis.

V1 deliberately requires one source with confirmed Revenue, Cost of Goods
Sold, and Currency mappings. It will not combine unrelated periods, sources,
or generic operating costs and pretend they form gross profit.
"""

from __future__ import annotations

import re
import uuid
from dataclasses import dataclass
from datetime import datetime
from decimal import Decimal, InvalidOperation
from typing import Any

from sqlalchemy import func, select
from sqlalchemy.ext.asyncio import AsyncSession

from app.db.models.browser import BrowserExtractionProfile
from app.db.models.dataset import DataSource, IngestionJob, StagedRow
from app.db.models.semantic import SemanticMapping

REVENUE_CONCEPT = "Finance.Revenue"
COGS_CONCEPT = "Finance.CostOfGoodsSold"
CURRENCY_CONCEPT = "Finance.Currency"
SUPPORTED_SNAPSHOT_MODES = frozenset({"append", "replace_active"})
_MAX_ROWS = 200_000
_CURRENCY_PATTERN = re.compile(r"^[A-Z]{3}$")


class BaselineUnavailable(Exception):
    """The source cannot support a reliable, explainable profit scenario."""

    def __init__(
        self,
        code: str,
        detail: str,
        *,
        missing: list[str] | None = None,
    ) -> None:
        super().__init__(detail)
        self.code = code
        self.detail = detail
        self.missing = missing or []


@dataclass(frozen=True)
class ProfitBaseline:
    source_id: uuid.UUID
    source_name: str
    snapshot_mode: str
    ingestion_job_ids: list[uuid.UUID]
    revenue_concept: str
    revenue_column: str
    cogs_concept: str
    cogs_column: str
    currency_concept: str
    currency_column: str
    currency: str
    revenue: Decimal
    cogs: Decimal
    rows_considered: int
    latest_ingested_at: datetime | None

    @property
    def gross_profit(self) -> Decimal:
        return self.revenue - self.cogs

    @property
    def gross_margin_pct(self) -> Decimal | None:
        if self.revenue == 0:
            return None
        return self.gross_profit / self.revenue * Decimal(100)

    def to_snapshot(self) -> dict[str, Any]:
        return {
            "source_id": str(self.source_id),

def _parse_number(raw: Any, *, field_name: str, row_number: int) -> Decimal:
    if raw is None or isinstance(raw, bool):
        raise BaselineUnavailable(
            "incomplete_data",
            f"row {row_number} has no usable {field_name} value; "
            "the calculation was blocked rather than silently omitting data",
        )
    try:
        value = Decimal(str(raw).strip().replace(",", ""))
    except (InvalidOperation, ValueError, TypeError) as exc:
        raise BaselineUnavailable(
            "invalid_data",
            f"row {row_number} has an invalid {field_name} value",
        ) from exc
    if not value.is_finite():
        raise BaselineUnavailable(
            "invalid_data", f"row {row_number} has a non-finite {field_name} value"
        )
    if value < 0:
        raise BaselineUnavailable(
            "negative_financial_value",
            f"row {row_number} has negative {field_name}; "
            "refund and credit treatment must be defined before inclusion",
        )
    return value


async def resolve_profit_baseline(
    db: AsyncSession,
    *,
    tenant_id: uuid.UUID,
    source_id: uuid.UUID,
) -> ProfitBaseline:
    source = (
        await db.execute(
            select(DataSource).where(
                DataSource.id == source_id,
                DataSource.tenant_id == tenant_id,
            )
        )
    ).scalar_one_or_none()
    if source is None:
        raise BaselineUnavailable("source_not_found", "source not found for this tenant")
    if not source.is_active:
        raise BaselineUnavailable("source_inactive", "the source is inactive")
    if source.snapshot_mode not in SUPPORTED_SNAPSHOT_MODES:
        raise BaselineUnavailable(
            "unsupported_snapshot_mode",
            f"snapshot mode {source.snapshot_mode!r} is not supported",
        )

    required = (REVENUE_CONCEPT, COGS_CONCEPT, CURRENCY_CONCEPT)
    mappings: dict[str, SemanticMapping] = {}
    missing: list[str] = []
    unconfirmed: list[str] = []
    for concept in required:
        confirmed = await _mapping(
            db,
            tenant_id=tenant_id,
            source_id=source.id,
            concept=concept,
            confirmed_only=True,
        )
        if confirmed is not None:
            mappings[concept] = confirmed
            continue
        any_mapping = await _mapping(
            db,
            tenant_id=tenant_id,
            source_id=source.id,
            concept=concept,
            confirmed_only=False,
        )
        (unconfirmed if any_mapping is not None else missing).append(concept)
    if missing or unconfirmed:
        unmet = sorted(set(missing + unconfirmed))
        raise BaselineUnavailable(
            "missing_mappings",
            "a reliable profit baseline requires confirmed mappings for: " + ", ".join(unmet),
            missing=unmet,
        )

    active_job_id = await _active_job_id(db, tenant_id=tenant_id, source=source)
    if source.snapshot_mode == "replace_active" and active_job_id is None:
        raise BaselineUnavailable(
            "no_active_snapshot",
            "the replace_active source has no active ingestion snapshot",
        )

    statement = (
        select(StagedRow.raw_data, StagedRow.row_number, StagedRow.job_id)
        .join(IngestionJob, IngestionJob.id == StagedRow.job_id)
        .where(
            StagedRow.tenant_id == tenant_id,
            StagedRow.source_id == source.id,
            IngestionJob.status == "succeeded",
        )
    )
    if active_job_id is not None:
        statement = statement.where(StagedRow.job_id == active_job_id)
    statement = statement.order_by(StagedRow.job_id, StagedRow.row_number).limit(_MAX_ROWS + 1)
    staged = (await db.execute(statement)).all()
    if len(staged) > _MAX_ROWS:
        raise BaselineUnavailable(
            "too_many_rows",
            f"the source exceeds the {_MAX_ROWS:,}-row v1 calculation limit",
        )
    if not staged:
        raise BaselineUnavailable("no_staged_rows", "the source has no successful staged rows")

    revenue_column = mappings[REVENUE_CONCEPT].source_column
    cogs_column = mappings[COGS_CONCEPT].source_column
    currency_column = mappings[CURRENCY_CONCEPT].source_column
    total_revenue = Decimal(0)
    total_cogs = Decimal(0)
    currencies: set[str] = set()
    job_ids: set[uuid.UUID] = set()
    for raw_data, row_number, job_id in staged:
        data = raw_data if isinstance(raw_data, dict) else {}
        row_label = int(row_number)
        revenue = _parse_number(
            data.get(revenue_column),
            field_name="revenue",
            row_number=row_label,
        )
        cogs = _parse_number(
            data.get(cogs_column),
            field_name="COGS",
            row_number=row_label,
        )
        raw_currency = data.get(currency_column)
        currency = str(raw_currency or "").strip().upper()
        if not _CURRENCY_PATTERN.fullmatch(currency):
            raise BaselineUnavailable(
                "unknown_currency",
                f"row {row_label} has no normalized three-letter currency code; "
                "currency must be confirmed before combining financial values",
                missing=[CURRENCY_CONCEPT],
            )
        currencies.add(currency)
        if len(currencies) > 1:
            raise BaselineUnavailable(
                "mixed_currencies",
                "the source contains multiple currencies; currency conversion "
                "is not modeled, so the values cannot be combined",
            )
        total_revenue += revenue
        total_cogs += cogs
        job_ids.add(job_id)

    latest_ingested_at = (
        await db.execute(
            select(func.max(IngestionJob.finished_at)).where(IngestionJob.id.in_(job_ids))
        )
    ).scalar_one_or_none()
    return ProfitBaseline(
        source_id=source.id,
        source_name=source.name,
        snapshot_mode=source.snapshot_mode,
        ingestion_job_ids=sorted(job_ids, key=str),
        revenue_concept=REVENUE_CONCEPT,
        revenue_column=revenue_column,
        cogs_concept=COGS_CONCEPT,
        cogs_column=cogs_column,
        currency_concept=CURRENCY_CONCEPT,
        currency_column=currency_column,
        currency=next(iter(currencies)),
        revenue=total_revenue,
        cogs=total_cogs,
        rows_considered=len(staged),
        latest_ingested_at=latest_ingested_at,
    )


def is_baseline_current(
    *, saved_snapshot: dict[str, Any], current_snapshot: dict[str, Any]
) -> bool:
    return saved_snapshot == current_snapshot

async def calculate(
    db: AsyncSession,
    *,
    tenant_id: uuid.UUID,
    source_id: uuid.UUID,
    assumptions: engine.ProfitAssumptions,
) -> tuple[dict[str, Any], ProfitBaseline, engine.ProfitSensitivityResult]:
    baseline = await resolve_profit_baseline(db, tenant_id=tenant_id, source_id=source_id)
    try:
        result = engine.simulate_profit(
            baseline=engine.ProfitBaseline(
                revenue=baseline.revenue,
                cogs=baseline.cogs,
                currency=baseline.currency,
            ),
            assumptions=assumptions,
        )
    except engine.ScenarioCalculationError as exc:
        raise ScenarioServiceError(str(exc)) from exc
    return _result_document(result), baseline, result


def _assumptions_from_scenario(scenario: WhatIfScenario) -> engine.ProfitAssumptions:

        )
    except (KeyError, TypeError, ArithmeticError, ValueError) as exc:

async def compare_scenarios(

async def mark_stale_scenarios(db: AsyncSession, *, tenant_id: uuid.UUID) -> int:
    scenarios = (
        (
            await db.execute(
                select(WhatIfScenario).where(
                    WhatIfScenario.tenant_id == tenant_id,
                    WhatIfScenario.status == "calculated",
                )
            )
        )
        .scalars()
        .all()
    )
    current_by_source: dict[uuid.UUID, dict[str, Any] | None] = {}
    updated = 0
    for scenario in scenarios:
        if scenario.source_id not in current_by_source:
            try:
                baseline = await resolve_profit_baseline(
                    db, tenant_id=tenant_id, source_id=scenario.source_id
                )
                current_by_source[scenario.source_id] = baseline.to_snapshot()
            except BaselineUnavailable:
                current_by_source[scenario.source_id] = None
        current = current_by_source[scenario.source_id]
        saved = scenario.baseline_snapshot or {}
        if current is None or not is_baseline_current(
            saved_snapshot=saved, current_snapshot=current
        ):
            scenario.status = "stale"
            scenario.stale_reason = (
                "source data changed or no longer supports a verified profit baseline"
            )
            updated += 1
    if updated:
        await db.flush()
    return updated

def to_read(scenario: WhatIfScenario) -> ScenarioRead:
    result: ScenarioResultRead | None = None
    provenance: ScenarioProvenanceRead | None = None
    supported = scenario.scenario_type == engine.SCENARIO_TYPE
    if supported:
        try:
            result = result_read(scenario.result or {})
            provenance = provenance_read(scenario.provenance or {})
        except (ArithmeticError, KeyError, TypeError, ValueError):
            supported = False

    return ScenarioRead(
        id=scenario.id,
        name=scenario.name,
        scenario_type=scenario.scenario_type,
        source_id=scenario.source_id,
        status=scenario.status if supported else "unsupported",
        assumptions=scenario.assumptions or {},
        result=result,
        provenance=provenance,
        is_stale=scenario.status == "stale",
        stale_reason=scenario.stale_reason,
        created_by_user_id=scenario.created_by_user_id,
        created_at=scenario.created_at,
        updated_at=scenario.updated_at,
        supported=supported,
        compatibility_note=(
            None
            if supported
            else "This saved scenario uses an unsupported format. Its historical "
            "values are preserved but cannot be represented as profit metrics."
        ),
    )

async def seed_canonical_concepts(db: AsyncSession) -> int:
    """
    Idempotent reconciliation of the global canonical concept catalog.

    Inserts concepts that don't exist and updates mutable fields on existing
    ones (so edits to this file are picked up on re-run). Never deletes.

    Returns the number of concepts reconciled (inserted or updated).
    """
    existing: dict[str, CanonicalConcept] = {
        c.key: c
        for c in (await db.execute(select(CanonicalConcept))).scalars().all()
    }

    reconciled = 0
    for key, display_name, domain, kind, value_type, synonyms, description in ALL_CONCEPTS:
        if key in existing:
            c = existing[key]
            c.display_name = display_name
            c.domain = domain
            c.kind = kind
            c.value_type = value_type
            c.synonyms = synonyms
            c.description = description
        else:
            db.add(CanonicalConcept(
                key=key,
                display_name=display_name,
                domain=domain,
                kind=kind,
                value_type=value_type,
                synonyms=synonyms,
                description=description,
            ))
        reconciled += 1

    await db.flush()
    return reconciled

async def require_scenario_csrf(request: Request) -> None:
    """Accept the standard CSRF header or the token carried by HTML forms."""
    cookie_token = request.cookies.get(settings.csrf_cookie_name)
    submitted_token = request.headers.get(settings.csrf_header_name)
    if not submitted_token:
        form = await request.form()
        token = form.get("csrf_token")
        submitted_token = token if isinstance(token, str) else None
    if (
        not cookie_token
        or not submitted_token
        or not hmac.compare_digest(cookie_token, submitted_token)
    ):
        raise HTTPException(status_code=403, detail="csrf check failed")


def _fmt_money(value: Decimal | str | None) -> str:
    """Group thousands, keep two decimals. Readability only, no rounding."""
    if value is None:
        return "(none)"
    if not isinstance(value, Decimal):
        value = Decimal(str(value))
    return f"{value:,.2f}"


def _fmt_pct(value: Decimal | str | None) -> str:
    if value is None:
        return "n/a"
    if not isinstance(value, Decimal):
        value = Decimal(str(value))
    sign = "+" if value > 0 else ""
    return f"{sign}{value}%"


templates.env.filters["money"] = _fmt_money
